cmake_minimum_required(VERSION 3.16)
project(retro_windows_launchers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(game asteroids pong snake)
  add_executable(${game}_saver WIN32 screensaver.cpp)
  target_compile_definitions(${game}_saver PRIVATE UNICODE _UNICODE RETRO_GAME=L"${game}")
  target_link_libraries(${game}_saver PRIVATE advapi32 user32)
  if(MINGW)
    target_link_options(${game}_saver PRIVATE -municode -static-libgcc -static-libstdc++)
  endif()
endforeach()
